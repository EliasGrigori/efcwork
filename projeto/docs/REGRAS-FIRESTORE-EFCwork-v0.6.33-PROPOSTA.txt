service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- quem é quem ----------
    function logado() { return request.auth != null; }
    function temPerfil() { return logado() && exists(/databases/$(database)/documents/sv2_users/$(request.auth.uid)); }
    function eu() { return get(/databases/$(database)/documents/sv2_users/$(request.auth.uid)).data; }
    function usuarioAtivo() { return temPerfil() && eu().get('bloqueado', false) != true; }
    function principal() { return usuarioAtivo() && eu().get('perfil', '') in ['principal', 'principal_temp']; }
    function liderOuPrincipal() { return principal() || (usuarioAtivo() && eu().get('estrela', false) == true); }
    function lanca() { return usuarioAtivo() && eu().get('perfil', '') != 'visitante'; }
    // ---------- ajudantes ----------
    function novo() { return request.resource.data; }
    function soMuda(campos) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(campos); }
    function meu(campo) { return request.resource.data.get(campo, '') == request.auth.uid; }
    function imagem(campo, max) { return !(campo in request.resource.data) || (request.resource.data[campo] is string && request.resource.data[campo].size() < max); }
    function imagemVazia(campo) { return request.resource.data.get(campo, 'x') == ''; }
    // ---------- v2.89.0: conta de teste ----------
    function contaTeste() { return eu().get('contaTeste', false) == true; }
    function testeNovo() { return !contaTeste() || novo().get('testeConta', false) == true; }
    function testeMuda() { return !contaTeste() || (resource.data.get('testeConta', false) == true && novo().get('testeConta', false) == true); }

    // ==================== APP DOCUMENTOS ANTIGO ====================
    // (16/10/2026: as regras publicadas não tinham bloco /appdata; o app Documentos antigo continua sem acesso, como já estava.)

    // ==================== INSTALAÇÃO E USUÁRIOS ====================
    match /sv2_meta/{id} {
      allow read: if true;                                           // a tela de entrar pergunta se já foi instalado
      allow create: if logado() && id == 'setup' && novo().get('por', '') == request.auth.uid;
      allow update, delete: if false;
    }
    match /sv2_users/{uid} {
      allow read: if usuarioAtivo();
      // 1º acesso: o próprio Principal (junto com sv2_meta/setup) | depois: só o Principal cria contas
      allow create: if (logado() && uid == request.auth.uid && novo().get('perfil', '') == 'principal'
                        && !exists(/databases/$(database)/documents/sv2_meta/setup))
                    || principal();
      // Principal muda tudo; cada um só o próprio cargo, status e tema
      // v2.74.0: cada um também completa o próprio cadastro (nome, e-mail, senha trocada)
      allow update: if principal()
                    || (usuarioAtivo() && uid == request.auth.uid && soMuda(['cargo', 'status', 'tema', 'nome', 'email', 'emailPendente', 'emailOk', 'precisaCompletar', 'completadoEm', 'senhaTrocadaEm']))
                    // v2.89.0: a conta de teste se bloqueia ao sair (só pode pôr bloqueado = true)
                    || (usuarioAtivo() && uid == request.auth.uid && resource.data.get('contaTeste', false) == true
                        && soMuda(['bloqueado', 'bloqueadoEm', 'motivoBloqueio']) && novo().bloqueado == true);
      allow delete: if false;
    }
    match /sv2_avatar/{uid} {
      allow read: if usuarioAtivo();
      allow create, update: if (usuarioAtivo() && uid == request.auth.uid && imagem('img', 60000))
                            || (principal() && imagemVazia('img'));
      allow delete: if false;
    }

    // ==================== AVISOS, AUDITORIA, AUTORIZAÇÕES ====================
    match /sv2_notifs/{id} {
      // PASSO 2 (v2.89.0): cada um lê só o que é para ele (o Principal lê tudo)
      allow read: if usuarioAtivo() && (principal()
                    || request.auth.uid in resource.data.dest
                    || 'todos' in resource.data.dest
                    || ('sup' in resource.data.dest && liderOuPrincipal()));
      allow create: if usuarioAtivo() && testeNovo() && novo().get('dest', null) is list && novo().dest.size() > 0;
      // v2.89.0: o Principal completa "dest" nos avisos antigos (só esse campo)
      allow update: if principal() && soMuda(['dest']) && novo().dest is list;
      // quem recebe marca só o próprio "lido"/"confirmado"
      allow update: if usuarioAtivo() && soMuda(['lidaPor', 'confirmadaPor'])
                    && novo().get('lidaPor', {}).diff(resource.data.get('lidaPor', {})).affectedKeys().hasOnly([request.auth.uid])
                    && novo().get('confirmadaPor', {}).diff(resource.data.get('confirmadaPor', {})).affectedKeys().hasOnly([request.auth.uid]);
      allow delete: if false;
    }
    match /sv2_audit/{id} {
      allow read: if principal();
      allow create: if (usuarioAtivo() && meu('userId')) && testeNovo();
      allow update, delete: if false;
    }
    match /sv2_aprov/{id} {
      allow read: if usuarioAtivo();
      allow create: if (usuarioAtivo() && meu('solicitanteId') && novo().get('status', '') == 'pendente') && testeNovo();
      // decide (v0.6.32, ver [R2]): o líder ★ indicado ou o Principal; só enquanto pendente; só status e decisão
      // [R2] v0.6.32: decide o líder ★ indicado no pedido (paraIds) que não é quem pediu, ou o Principal.
      // Antes: qualquer líder ★, inclusive no próprio pedido, ou qualquer pessoa posta em paraIds por quem pediu.
      allow update: if (usuarioAtivo() && resource.data.status == 'pendente'
                    && ((liderOuPrincipal() && request.auth.uid in resource.data.get('paraIds', []) && request.auth.uid != resource.data.get('solicitanteId', '')) || principal())
                    && soMuda(['status', 'decisao']) && novo().decisao.por == request.auth.uid) && testeMuda();
      allow delete: if false;
    }
    match /sv2_resumos/{id} {
      allow read, create, update: if principal();
      allow delete: if false;
    }
    match /sv2_chat/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && meu('porId') && novo().texto is string && novo().texto.size() > 0 && novo().texto.size() <= 1000) && testeNovo();
      allow update, delete: if false;
    }

    // ==================== CONFIGURAÇÕES ====================
    match /sv2_cfg/{doc} {
      allow read: if usuarioAtivo();
      allow create: if ((doc in ['doccontrole', 'backup', 'arquivo'] && principal())
                            || (!(doc in ['doccontrole', 'backup', 'arquivo']) && liderOuPrincipal())
                            || (doc == 'almox' && lanca() && soMuda(['kits', 'atualizadoEm', 'atualizadoPor', 'srv']))) && testeNovo();
      allow update: if ((doc in ['doccontrole', 'backup', 'arquivo'] && principal())
                            || (!(doc in ['doccontrole', 'backup', 'arquivo']) && liderOuPrincipal())
                            || (doc == 'almox' && lanca() && soMuda(['kits', 'atualizadoEm', 'atualizadoPor', 'srv']))) && testeMuda();
      allow delete: if false;
    }

    // ==================== DOCUMENTOS ====================
    match /sv2_docs/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_caixas/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_lanc/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && meu('lancadoPor')) && testeNovo();
      // correção de lançamento (quem decide o pedido pode não ser ★)
      allow update: if (lanca() && soMuda(['data', 'obs', 'alterado', 'historico', 'ultimaAprovId'])) && testeMuda();
      allow delete: if false;
    }

    // ==================== CLORO E BEBEDOUROS ====================
    match /sv2_bebs/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca()) && testeNovo();
      allow update: if (liderOuPrincipal() || (lanca() && soMuda(['codigo', 'local', 'situacao', 'descricao', 'aprovId']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_cloro/{id} { allow read: if usuarioAtivo(); allow create: if (lanca()) && testeNovo();
      allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_cloro_falta/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_cloro_foto/{id} {
      allow read: if usuarioAtivo();
      allow create: if ((lanca() && imagem('img', 400000)) || (liderOuPrincipal() && imagemVazia('img'))) && testeNovo();
      allow update: if (liderOuPrincipal() && imagemVazia('img')) && testeMuda();       // eliminar a imagem (24 h depois de "confere")
      allow delete: if false;
    }

    // ==================== COLETAS, ÁGUA, CADASTROS ====================
    match /sv2_talhoes/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_analises/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_etqmodelos/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_prog/{id} { allow read: if usuarioAtivo(); allow create: if (lanca() && meu('criadoPor')) && testeNovo(); allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_coletas/{id} { allow read: if usuarioAtivo(); allow create: if (lanca() && meu('criadoPor')) && testeNovo(); allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_aguacol/{id} { allow read: if usuarioAtivo(); allow create: if (lanca() && meu('criadoPor')) && testeNovo(); allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_rodizio/{id} { allow read: if usuarioAtivo(); allow create: if (lanca()) && testeNovo();
      allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_aguaproj/{id} { allow read: if usuarioAtivo(); allow create: if (lanca()) && testeNovo();
      allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_aguaponto/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_agua_foto/{id} {
      allow read: if usuarioAtivo();
      allow create: if ((lanca() && imagem('dataUrl', 400000)) || (liderOuPrincipal() && imagemVazia('dataUrl'))) && testeNovo();
      allow update: if (liderOuPrincipal() && imagemVazia('dataUrl')) && testeMuda();   // eliminar 7 dias depois da conclusão
      allow delete: if false;
    }
    match /sv2_proc/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal() || (lanca() && soMuda(['pedidos']))) && testeNovo();
      allow update: if (liderOuPrincipal() || (lanca() && soMuda(['pedidos']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_proc_pag/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal() && imagem('dataUrl', 1000000)) && testeNovo();
      allow update: if (liderOuPrincipal() && imagemVazia('dataUrl')) && testeMuda();
      allow delete: if false;
    }

    // ==================== APONTAMENTOS E AUDITORIA INTERNA ====================
    match /sv2_apont/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && meu('porId')) && testeNovo();
      allow update: if (liderOuPrincipal()
                    || (lanca() && resource.data.get('porId', '') == request.auth.uid && soMuda(['origem', 'auditoria', 'hist']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_fotos/{id} {
      allow read: if usuarioAtivo();
      allow create: if ((lanca() && imagem('dataUrl', 400000)) || (liderOuPrincipal() && imagemVazia('dataUrl'))) && testeNovo();
      allow update: if (liderOuPrincipal() && imagemVazia('dataUrl')) && testeMuda();   // apagar 15 dias depois de resolvido (LGPD)
      allow delete: if false;
    }
    match /sv2_rnc/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()
                    || (usuarioAtivo() && resource.data.get('auditorId', '') == request.auth.uid
                        && novo().get('auditorId', '') == resource.data.get('auditorId', ''))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_rnc_foto/{id} {
      allow read: if usuarioAtivo();
      allow create: if (usuarioAtivo() && imagem('dataUrl', 400000)) && testeNovo();
      // v2.102.42: Lixeira da RNC — com 15 dias o líder ★/Principal só pode ESVAZIAR a foto (dataUrl e base vazios).
      allow update: if (liderOuPrincipal() && imagemVazia('dataUrl') && request.resource.data.get('base', '') == ''
                    && soMuda(['dataUrl', 'base', 'apagadaEm'])) && testeMuda();
      allow delete: if false;
    }
    // v2.102.13: análise de cada NC (uma linha por NC, id = id da NC). Só o responsável pela resposta escreve;
    // o líder ★/Principal só troca o responsável (e pode criar a linha já com o novo responsável). Nada se apaga.
    match /sv2_rnc_plano/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && (meu('respId') || liderOuPrincipal())) && testeNovo();
      allow update: if (lanca() && (resource.data.get('respId', '') == request.auth.uid
                    || (liderOuPrincipal() && soMuda(['respId', 'respNome', 'respHist', 'rev', 'em'])))) && testeMuda();
      allow delete: if false;
    }
    // v2.102.12: prioridade GUT das NCs (uma linha por NC, id = id da NC). Cria quem lança (não visitante), como dono da nota;
    // muda o dono da nota ou o líder ★/Principal, sem trocar o dono. Nada se apaga (tirar a nota = cancelada).
    match /sv2_rnc_gut/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && meu('porId')) && testeNovo();
      allow update: if (lanca() && (resource.data.get('porId', '') == request.auth.uid || liderOuPrincipal())
                    && novo().get('porId', '') == resource.data.get('porId', '')) && testeMuda();
      allow delete: if false;
    }
    // v2.102.20: Análise de problemas (ferramentas da qualidade fora da RNC). Abre o líder ★ ou o Principal (como dono da abertura).
    // Escreve a análise e as ações só o responsável pela resposta, sem trocar o responsável nem a situação;
    // o líder ★/Principal troca o responsável, confere a eficácia, conclui e cancela. Nada se apaga.
    match /sv2_analise/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && liderOuPrincipal() && meu('abertoPorId')) && testeNovo();
      allow update: if (lanca() && (liderOuPrincipal()
                    || (resource.data.get('respId', '') == request.auth.uid
                        && novo().get('respId', '') == resource.data.get('respId', '')
                        && novo().get('situacao', '') == resource.data.get('situacao', '')))) && testeMuda();
      allow delete: if false;
    }

    // ==================== ALMOXARIFADO ====================
    // [R3] v0.6.32: cadastrar e alterar material é do líder ★ ou do Principal (o pedido do colaborador é aplicado por quem aprova);
    // quem lança só atualiza o preço, na Entrada.
    match /sv2_alm_itens/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal() || (lanca() && soMuda(['preco']) && novo().preco is number && novo().preco >= 0)) && testeMuda(); allow delete: if false; }
    match /sv2_alm_mov/{id} { allow read: if usuarioAtivo(); allow create: if (lanca()) && testeNovo();
      allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_alm_sol/{id} { allow read: if usuarioAtivo(); allow create: if (lanca()) && testeNovo();
      allow update: if (lanca()) && testeMuda(); allow delete: if false; }
    match /sv2_alm_pref/{uid} {
      allow read: if usuarioAtivo() && (uid == request.auth.uid || principal());
      allow create, update: if usuarioAtivo() && uid == request.auth.uid;
      allow delete: if false;
    }
    // v2.62.0 — Conquistas: um documento por pessoa (cada um grava só o seu) e o 'equipe' (qualquer usuário ativo que
    // não seja visitante recalcula 1×/dia). Todos leem (sem ranking). Nada se apaga.
    match /sv2_conquistas/{id} {
      allow read: if usuarioAtivo();
      allow create, update: if lanca() && !contaTeste() && (id == request.auth.uid || id == 'equipe');
      allow delete: if false;
    }
    match /sv2_alm_equip/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca()) && testeNovo();
      allow update: if (liderOuPrincipal()
                    || (lanca() && soMuda(['situacao', 'local', 'posicao', 'guardadoEm', 'sitAntes', 'posAntes', 'restricaoCal',
                                           'qtdLocal', 'qtdGuard', 'trfQtd', 'transito', 'hist', 'atualizadoEm']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_alm_equip_foto/{id} {
      allow read: if usuarioAtivo();
      allow create: if ((lanca() && imagem('dataUrl', 400000)) || (liderOuPrincipal() && imagemVazia('dataUrl'))) && testeNovo();
      allow update: if (liderOuPrincipal() && imagemVazia('dataUrl')) && testeMuda();
      allow delete: if false;
    }
    match /sv2_alm_cal/{id} { allow read: if usuarioAtivo(); allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda(); allow delete: if false; }
    match /sv2_alm_inv/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal() || (lanca() && soMuda(['contagens', 'folhas']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_alm_transf/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && novo().envio.porId == request.auth.uid) && testeNovo();
      allow update: if (liderOuPrincipal()
                    || (lanca() && soMuda(['receb', 'situacao', 'atualizadoEm', 'canceladaEm', 'canceladaPor', 'canceladaMotivo']))) && testeMuda();
      allow delete: if false;
    }
    match /sv2_alm_foto/{id} {

      allow delete: if false;
    }
    // v2.70.0 — S10 · Normas e editais da auditoria interna. Líder ★/Principal cadastra e altera (desativar/tirar com motivo
    // é um update); todos os usuários ativos leem (o checklist aparece na RNC). Nada se apaga.
    match /sv2_normas/{id} {

      allow update: if (liderOuPrincipal()) && testeMuda();

    }
    match /sv2_normas_ed/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal()) && testeNovo();
      allow update: if (liderOuPrincipal()) && testeMuda();
      allow delete: if false;
    }
    // v2.74.0 — Login → e-mail (para entrar com o login e recuperar a senha). Qualquer um LÊ um documento pelo login exato
    // (precisa antes de entrar), ninguém lista; só o Principal ou o dono do login grava. Nada se apaga.
    match /sv2_login/{login} {
      allow get: if true;
      allow list: if principal();
      // [R1] v0.6.32: só quem tem conta ativa, e só o PRÓPRIO login (o id do documento = o login gravado no cadastro dele).
      // Antes bastava estar logado (até o login anônimo do EFCQuiz) e dava para ligar o login de um colega ao próprio e-mail.
      allow create: if principal() || (usuarioAtivo() && login == eu().get('login', '') && novo().get('uid', '') == request.auth.uid);
      allow update: if principal() || (usuarioAtivo() && login == eu().get('login', '') && resource.data.get('uid', '') == request.auth.uid && novo().get('uid', '') == request.auth.uid);
      allow delete: if false;
    }
    // v2.74.0 — Pedidos de recuperação de senha: a tela de login (sem estar logado) só CRIA um pedido pequeno (e-mail e
    // horário); só o Principal lê e marca como visto. Nada se apaga.
    match /sv2_recup/{id} {
      allow create: if novo().keys().hasOnly(['email', 'ts', 'srv']) && novo().email is string && novo().email.size() < 120
                    && novo().ts is string && novo().ts.size() < 40;
      allow read, update: if principal();
      allow delete: if false;
    }
    // v2.80.0 — Bugs e sugestões: cria em nome dele, sempre como "novo"; só o Principal muda a situação e a prioridade.
    // EFCwork v0.6.20: cada um LÊ só o que lançou (o Principal lê tudo); quem reportou só acrescenta comentário. Nada se apaga.
    match /sv2_bugs/{id} {
      allow read: if principal() || (usuarioAtivo() && resource.data.get('porId', '') == request.auth.uid);
      allow create: if (usuarioAtivo() && meu('porId') && novo().get('status', '') == 'novo') && testeNovo();
      allow update: if (principal()
                    || (usuarioAtivo() && resource.data.get('porId', '') == request.auth.uid && soMuda(['coment']))) && testeMuda();
      allow delete: if false;
    }
    // v2.92.0 — Foto do bug à parte (sv2_bugs_foto/{id do bug}); EFCwork v0.6.20: também a imagem de cada comentário
    // ({id do bug}_c…, com donoId = quem reportou). Lê quem enviou, quem reportou e o Principal; ninguém altera. Nada se apaga.
    match /sv2_bugs_foto/{id} {
      allow read: if principal() || (usuarioAtivo() && (resource.data.get('porId', '') == request.auth.uid
                                                      || resource.data.get('donoId', '') == request.auth.uid));
      allow create: if (usuarioAtivo() && meu('porId') && novo().dataUrl is string && imagem('dataUrl', 400000)) && testeNovo();
      allow update, delete: if false;
    }
    // EFCwork v0.6.20 — Contador do número do relato (sv2_bugs_seq/n = { ultimo }): todo usuário ativo lê e só faz o número SUBIR
    // (o app usa uma transação). Quem não é Principal não vê os outros relatos, por isso o número vem daqui. Nada se apaga.
    match /sv2_bugs_seq/{id} {
      allow read: if usuarioAtivo();
      // [R5] v0.6.32: o Principal acerta o contador pelo maior número que vê; os outros só sobem de 1 a 3 por vez
      allow create, update: if usuarioAtivo() && id == 'n' && novo().keys().hasOnly(['ultimo', 'ts', 'porId']) && novo().get('ultimo', 0) is int
                            && (resource == null || novo().ultimo > resource.data.get('ultimo', 0))
                            && (principal() || novo().ultimo <= (resource == null ? 0 : resource.data.get('ultimo', 0)) + 3);
      allow delete: if false;
    }
    // v2.81.0 — Tarefas (v0.6.32: ver [R4]): todo usuário ativo lê; cria a tarefa para ele mesmo, ou o líder ★/Principal para outra pessoa
    // (a repetição cria a próxima); altera quem é o responsável, quem passou, os superiores do responsável
    // (gravados em "lideres") ou o Principal. Nada se apaga (cancelar com motivo).
    match /sv2_tarefas/{id} {
      allow read: if usuarioAtivo();
      // [R4] v0.6.32: para si mesmo, todos; para outra pessoa, só o líder ★ ou o Principal (gravando porId = ele)
      allow create: if (usuarioAtivo() && (meu('respId') || (meu('porId') && liderOuPrincipal()))) && testeNovo();
      allow update: if (usuarioAtivo() && (request.auth.uid == resource.data.get('respId', '') || request.auth.uid == resource.data.get('porId', '')
                    || request.auth.uid in resource.data.get('lideres', []) || principal())
                    // [R4] v0.6.32: trocar o responsável, quem passou ou os superiores da tarefa, só o líder ★ ou o Principal
                    && (liderOuPrincipal() || (novo().get('respId', '') == resource.data.get('respId', '')
                        && novo().get('porId', '') == resource.data.get('porId', '') && novo().get('lideres', []) == resource.data.get('lideres', [])))) && testeMuda();
      allow delete: if false;
    }

    // ==================== EFCwork v0.6.9 / v0.6.10 · IMPRESSOS E KIT DO LABORATÓRIO ====================
    // efd_docs: ficha de cada impresso (PDF, Word, Excel...). Todos os usuários ativos leem; só líder ★/Principal grava. Nada se apaga.
    match /efd_docs/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal() && novo().get('v', 0) == 1 && novo().get('id', '') == id) && testeNovo();
      allow update: if (liderOuPrincipal() && novo().get('v', 0) == 1 && novo().get('id', '') == id) && testeMuda();

    // efd_partes: o arquivo em pedaços de até 900.000 caracteres (versão antiga e lixeira gravam b64 vazio).
    match /efd_partes/{id} {
      allow read: if usuarioAtivo();
      allow create: if (liderOuPrincipal() && novo().get('b64', '') is string && novo().get('b64', '').size() <= 900000) && testeNovo();
      allow update: if (liderOuPrincipal() && novo().get('b64', '') is string && novo().get('b64', '').size() <= 900000) && testeMuda();

    // sv2_impressos: o kit (papel ligado ao laboratório). Todos leem; líder ★/Principal monta. Nada se apaga.
    match /sv2_impressos/{id} {
      allow read: if usuarioAtivo();

      allow update: if (liderOuPrincipal() && novo().get('v', 0) == 1) && testeMuda();

    // sv2_envio_imp: o que saiu em cada remessa (para reimprimir). Quem lança coletas grava, sempre em nome dele. Nada se apaga.
    match /sv2_envio_imp/{id} {

    // [R6] sv2_rem_docs (v0.6.33): (o anexo é conferido com getAfter: arquivo e registro vão no mesmo lote) Documentos da remessa — o que saiu de cada papel (impresso, preenchido no Vértice) e a via assinada anexada.
    // Quem lança coletas grava, sempre em nome dele. Nada se apaga: tirar um anexo é marcar removido (quem anexou, o líder ★ ou o Principal).
    match /sv2_rem_docs/{id} {
      allow read: if usuarioAtivo();
      allow create: if (lanca() && meu('porId') && novo().get('removido', true) == false
                        && novo().keys().hasOnly(['v', 'k', 'item', 'nome', 'tipo', 'porId', 'porNome', 'em', 'removido', 'docId', 'rev', 'paginas', 'copias', 'dadosJson', 'arqId', 'bytes', 'srv', 'testeConta'])
                        && novo().get('tipo', '') in ['impresso', 'aberto', 'preenchido', 'foto', 'pdf']
                        && novo().get('k', '') is string && novo().get('k', '').size() > 0 && novo().get('k', '').size() <= 200
                        && novo().get('item', '') is string && novo().get('item', '').size() > 0 && novo().get('item', '').size() <= 80
                        && novo().get('nome', '') is string && novo().get('nome', '').size() <= 160
                        && novo().get('porNome', '') == eu().get('nome', '')
                        && novo().get('dadosJson', '') is string && novo().get('dadosJson', '').size() <= 200000
                        && (!(novo().get('tipo', '') in ['foto', 'pdf'])
                            || getAfter(/databases/$(database)/documents/sv2_rem_arq/$(novo().get('arqId', 'x'))).data.get('porId', '') == request.auth.uid)) && testeNovo();

    // ==================== EFCQuiz v2.2.0 (celulares de verdade) ====================
    // Responsável = Principal ou líder ★ com conta ativa (a mesma do EFClaudian). A turma entra com login ANÔNIMO
    // do Firebase: só lê a sessão da chave que digitou e só cria a própria entrada e a própria resposta, uma vez.
    // Usuário anônimo não tem sv2_users, então não lê nem grava nada das coleções sv2_* (usuarioAtivo() = falso).
    // Nada se apaga (allow delete: if false).
    function efqSess(sid) { return get(/databases/$(database)/documents/efq_sessoes/$(sid)).data; }
    function efqDono(sid) { return liderOuPrincipal() && efqSess(sid).dono == request.auth.uid; }
    function efqAberta(sid) { return efqSess(sid).status != 'encerrada'; }
    function efqTexto(campo, max) { return !(campo in request.resource.data) || (request.resource.data[campo] is string && request.resource.data[campo].size() <= max); }
    match /efq_chaves/{chave} {

        allow list: if efqDono(sid);
        allow create: if logado() && novo().pid == request.auth.uid && rid == request.auth.uid + '__' + novo().item
                      && efqAberta(sid) && exists(/databases/$(database)/documents/efq_sessoes/$(sid)/part/$(request.auth.uid));

      }
    }
    // mensagens do perfil (pergunta, sugestão, reclamação): quem mandou e o responsável da sessão leem;
    // só o responsável responde, marca como lida ou esconde
    match /efq_msgs/{id} {

      allow delete: if false;
    }
